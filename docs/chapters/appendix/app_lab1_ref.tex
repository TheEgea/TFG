\section*{Lab Startup Checklist}

\begin{enumerate}
    \item Start all nodes from the EVE-NG web interface in order:
          pfSense $\to$ VyOS $\to$ Server $\to$ PC1 $\to$ Parrot.
    \item Verify the bridge interfaces on the EVE-NG host are active (persistent via
          \texttt{udev} rule \texttt{99-lab1-bridges.rules}):
\begin{lstlisting}[language=bash]
ip addr show vnet0_2   # Should show 192.168.20.254/24
ip addr show vnet0_3   # Should show 192.168.10.254/24
ip route show 172.16.0.0/30  # Should route via 192.168.20.1
\end{lstlisting}
    \item If bridges are missing (e.g.\ after EVE-NG host reboot), restore manually:
\begin{lstlisting}[language=bash]
ip addr add 192.168.20.254/24 dev vnet0_2
ip addr add 192.168.10.254/24 dev vnet0_3
ip route add 172.16.0.0/30 via 192.168.20.1
\end{lstlisting}
    \item Verify DNS resolution from Parrot (DNS must point to 192.168.0.29):
\begin{lstlisting}[language=bash]
# On Parrot:
curl http://lab1          # Should return PEBCAK Corp HTML
\end{lstlisting}
    \item Confirm all flags are in place:
\begin{lstlisting}[language=bash]
# Flag 1 (Server)
sshpass -p '!Bl4kM3s$' ssh blackmesa@192.168.20.50 cat ~/flag.txt
# Expected: FLAG{p3bc4k_s3rv3r_0wn3d}

# Flag 2 (pfSense) -- from EVE-NG host
sshpass -p 'pfsense' ssh -o PubkeyAuthentication=no \
  -o PreferredAuthentications=password admin@172.16.0.1 cat /root/flag.txt
# Expected: FLAG{d3f4ult_cr3ds_k1ll_s3cur1ty}

# Flag 3 (PC1) -- via Server
ssh blackmesa@192.168.20.50 "ssh skynet@192.168.10.50 cat ~/flag.txt"
# Expected: FLAG{sk1n3t_b3c4m3_s3nt13nt}
\end{lstlisting}
\end{enumerate}
