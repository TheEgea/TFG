\subsection{A2 -- AES-ECB Oracle Attack}

\textbf{Flag:} \flag{H4U\{4es\_3cb\_bl0ck\_4tt4ck\}}

\textbf{Technique:} AES-ECB byte-at-a-time oracle.

AES-ECB encrypts each 16-byte block independently. If the oracle encrypts
\texttt{attacker\_input + secret}, the attacker can align the unknown bytes
one at a time at the end of a block, then brute-force each byte:

\begin{lstlisting}[language=python]
# Pseudocode for byte-at-a-time ECB attack
recovered = b""
for i in range(len(secret)):
    block_index = i // 16
    padding_len = 15 - (i % 16)
    padding = b"A" * padding_len
    target_block = oracle(padding)[block_index*16:(block_index+1)*16]
    for byte in range(256):
        candidate = oracle(padding + recovered + bytes([byte]))
        if candidate[block_index*16:(block_index+1)*16] == target_block:
            recovered += bytes([byte])
            break
\end{lstlisting}

\begin{mdframed}[backgroundcolor=labgray, linewidth=0.5pt]
\textbf{Teaching point:} ECB mode is deterministic and stateless. Identical
plaintext blocks always produce identical ciphertext blocks. This makes it
vulnerable to chosen-plaintext attacks in oracle settings. Use CBC or GCM in production.
\end{mdframed}

\subsection{A3 -- Vigenere Cipher + Kasiski Examination}

\textbf{Flag:} \flag{H4U\{v1g3n3r3\_k4s1sk1\_4tt4ck\}}\\
\textbf{Key:} \texttt{NEXATECH}

\textbf{Solution steps:}
\begin{enumerate}[leftmargin=2em]
  \item Find repeated trigrams and their distances. The GCD of distances is 8,
        indicating a key length of 8.
  \item Split the ciphertext into 8 interleaved streams.
  \item Apply frequency analysis to each stream as a Caesar cipher.
  \item The most frequent letter in each stream corresponds to `E' in English.
\end{enumerate}

\begin{lstlisting}[language=python]
from itertools import cycle

def vigenere_decrypt(ciphertext, key):
    result = []
    key_cycle = cycle(key.upper())
    for char in ciphertext:
        if char.isalpha():
            shift = ord(next(key_cycle)) - ord('A')
            base = ord('A') if char.isupper() else ord('a')
            result.append(chr((ord(char) - base - shift) % 26 + base))
        else:
            result.append(char)
    return ''.join(result)

plaintext = vigenere_decrypt(open('A3_documento_clasificado.txt').read(), 'NEXATECH')
\end{lstlisting}

\subsection{A4 -- XOR Repeating Key}

\textbf{Flag:} \flag{H4U\{x0r\_r3p34t1ng\_k3y\}}

\textbf{Solution:}
\begin{lstlisting}[language=python]
ciphertext = bytes.fromhex(open('A4_mensaje_xor.hex').read().strip())

# Crib-dragging: try known word 'the' at every position
crib = b'the'
for i in range(len(ciphertext) - len(crib)):
    key_fragment = bytes(c ^ p for c, p in zip(ciphertext[i:], crib))
    # key_fragment[0] corresponds to key[i % key_len]

# Once key length is determined (use index of coincidence), recover full key
key_len = 8  # determined by IC analysis
key = bytearray()
for i in range(key_len):
    stream = ciphertext[i::key_len]
    # frequency analysis: most frequent byte XOR ord('e') = key byte
    freq = max(range(256), key=lambda k: sum(1 for b in stream if b == k))
    key.append(freq ^ ord('e'))

plaintext = bytes(b ^ key[i % len(key)] for i, b in enumerate(ciphertext))
\end{lstlisting}

\subsection{A5 -- RSA Small Exponent ($e = 3$, Cube Root Attack)}

\textbf{Flag:} \flag{H4U\{rsa\_sm4ll\_3xp0n3nt\}}

\textbf{Vulnerability:} When $e = 3$ and $m^3 < n$, the encryption $c = m^3 \bmod n$
performs no actual modular reduction. Therefore $c = m^3$ as an integer, and
$m = \sqrt[3]{c}$ exactly.

\begin{lstlisting}[language=python]
import gmpy2

# Read from A5_rsa_challenge.txt
n = int(...)   # modulus
e = 3
c = int(...)   # ciphertext

m, exact = gmpy2.iroot(c, e)
assert exact, "Cube root is not exact -- modular reduction occurred"

flag = m.to_bytes((m.bit_length() + 7) // 8, 'big').decode()
print(flag)
\end{lstlisting}

\begin{mdframed}[backgroundcolor=labgray, linewidth=0.5pt]
\textbf{Teaching point:} RSA requires a sufficiently large exponent (typically $e = 65537$)
or message padding (OAEP) to prevent small-exponent attacks. Textbook RSA with $e = 3$
and no padding is broken when the plaintext is small relative to the modulus.
\end{mdframed}

\subsection{C3 -- SHA-1 Hash Length Extension Attack}

\textbf{Flag:} \flag{H4U\{l3ngth\_3xt3ns10n\_sh4\}}

\textbf{Vulnerability:} The server computes \texttt{MAC = SHA1(secret + message)}.
SHA-1 (like MD5) uses Merkle-Damgard construction, which allows an attacker who
knows the MAC to append data and compute a valid MAC for the extended message
without knowing the secret.

\begin{lstlisting}[language=python]
import hlextend

# From C3_api_token.txt:
original_msg = b"user=analyst&role=viewer"
original_mac = "a3f......"  # 40-hex SHA1 from the file
secret_len = 16             # hinted in the challenge

sha = hlextend.new('sha1')
# Forge: extend with &access=admin
forged_msg = sha.extend(b'&access=admin', original_msg, secret_len, original_mac)
forged_mac = sha.hexdigest()

# Submit to verificator
# python3 C3_verificador.py <forged_msg_hex> <forged_mac>
import subprocess
result = subprocess.run(
    ['python3', 'C3_verificador.py', forged_msg.hex(), forged_mac],
    capture_output=True, text=True
)
print(result.stdout)  # H4U{l3ngth_3xt3ns10n_sh4}
\end{lstlisting}

\begin{mdframed}[backgroundcolor=labgray, linewidth=0.5pt]
\textbf{Teaching point:} \texttt{SHA1(secret + message)} is not a secure MAC.
Use HMAC-SHA256 instead. HMAC wraps the hash with two nested applications that
prevent length extension by design.
\end{mdframed}

\subsection{C4 -- ECDSA Nonce Reuse (secp256k1)}

\textbf{Flag:} \flag{H4U\{3cc\_n0nc3\_r3us3\_pwn3d\}}

\textbf{Vulnerability:} ECDSA security requires a unique random nonce $k$ for
every signature. If the same $k$ is reused across two signatures $(r, s_1, h_1)$
and $(r, s_2, h_2)$ (note: same $r$ implies same $k$), the private key is
algebraically recoverable.

\textbf{Recovery equations:}
\[
k = \frac{h_1 - h_2}{s_1 - s_2} \pmod{n}
\qquad
d = \frac{s_1 \cdot k - h_1}{r} \pmod{n}
\]

\begin{lstlisting}[language=python]
from sympy import mod_inverse

# secp256k1 curve order
n = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141

# Read from C4_ecdsa_signatures.txt -- find two entries with same r
# (r, s1, h1) and (r, s2, h2) share the same k

def modinv(a, m):
    return pow(a, -1, m)

k = (h1 - h2) * modinv(s1 - s2, n) % n
d = (s1 * k - h1) * modinv(r, n) % n

print(f"Recovered private key: {hex(d)}")

# Sign the challenge message with d
# Submit r_hex s_hex to C4_verificador.py
\end{lstlisting}

\begin{mdframed}[backgroundcolor=labgray, linewidth=0.5pt]
\textbf{Real-world precedent:} The Sony PlayStation 3 root key was recovered
in 2010 using exactly this attack. Sony's ECDSA implementation used a constant
$k$ for all firmware signatures.
\end{mdframed}
